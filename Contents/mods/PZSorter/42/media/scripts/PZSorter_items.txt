   WorldObjectSprite is the east face. The tile faces share GroupName
   Sorter and CustomName Crate, with Facing E, S, W, and N, so the
   place cursor can rotate. Weight is what a CustomItem pickup uses.
   Icon is the build-menu texture, which is a real file. "default" is not.
*/

module PZSorter
{
    item SorterCrate
    {
        DisplayCategory = Furniture,
        DisplayName = Sorter Crate,
        ItemType = base:moveable,
        Weight = 15,
        Icon = Build_PZSorterCrate,
        WorldObjectSprite = pzsorter_crate_01_0,
    }
}
